'use strict';
let img = true;
var huan = false
export function cursorClick(event) {
	if(huan == true){
		huan = false
	}else{
		huan = true
	}
}
export function update(value) {
	if(img==false){
		huan=false;
	}
	if(huan == false){
		thisScene.getLayer('22222').visible = false
		thisScene.getLayer('11111').visible = false
		thisScene.getLayer('33333').visible = false
		thisScene.getLayer('liangdu').visible = false
		thisScene.getLayer('555551').visible = false
		thisScene.getLayer('555552').visible = false
	}else if(huan == true){
		thisScene.getLayer('22222').visible = true
		thisScene.getLayer('11111').visible = true
		thisScene.getLayer('33333').visible = true
		thisScene.getLayer('liangdu').visible = true
		thisScene.getLayer('555551').visible = true
		thisScene.getLayer('555552').visible = true
	}
	return value;
}
export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('newproperty')){
		img=changedUserProperties.newproperty;
	}
}
