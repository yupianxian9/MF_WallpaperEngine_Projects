'use strict';
var audioBuffer = engine.registerAudioBuffers(16);
var state
var timer = 0

shared.sj = false

export function update(value) {
	timer += engine.frametime;
    state = !!audioBuffer.average.reduce((a, b) => a + b);
    if(timer > 1){
		thisScene.getLayer('圆角音频球').visible = state
		timer = 0
	}
	thisScene.getLayer('音频识别跳动').visible = state
	return value;
}

var speed = 0
var time = 0, intervalTime = 0, oldTime = 0, n = 0 , values = 0;
var ofg = 0

export function mediaPlaybackChanged(event) {
    if(event.state == 0){
		ofg = 0
	}else if(event.state == 1){
		ofg = 1
	}else if(event.state == 2){
		ofg = 2
	}else{
		ofg = 3
	}
}
//关闭 0    播放1   暂停2  
export function update(value) {
    intervalTime += engine.frametime;
	n++;
	if (time - oldTime > 500){
		let fps = 1 / (intervalTime / n);
		values = fps.toFixed(0);
		oldTime = time;
		intervalTime = n = 0;
	}
	// time += engine.frametime * 1000;
	if(values<80){
		speed = scriptProperties.speed/100;
	}else if(values<160){
		speed = scriptProperties.speed/200;
	}else if(values<230){
		speed = scriptProperties.speed/300;
	}else{
		speed = scriptProperties.speed/400;
	} 


    if(ofg == 0 || ofg == 2){
        value = new Vec3(
			WEMath.mix(value.x, scriptProperties.bX, speed), 
			value.y,
			value.z
		);
		
	}else if(ofg == 1){
        value = new Vec3(
			WEMath.mix(value.x, scriptProperties.aX, speed), 
			value.y,
			value.z,
		);
        
	}
    return value;
}
