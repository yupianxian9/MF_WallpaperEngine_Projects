```js
'use strict';

shared = {
	back: 1 
}
export function update(value) {

	if(shared.back == 1){
			thisScene.getLayer('00').visible = true
			thisScene.getLayer('纯色').visible = true
			thisScene.getLayer('可调整组合层').visible = true
			thisScene.getLayer('1').visible = true
			thisScene.getLayer('2').visible = true
			thisScene.getLayer('3').visible = true

			thisScene.getLayer('红').visible = false
			thisScene.getLayer('白').visible = false

	}else if(shared.back == 2){
			thisScene.getLayer('00').visible = false
			thisScene.getLayer('纯色').visible = false
			thisScene.getLayer('可调整组合层').visible = false
			thisScene.getLayer('1').visible = false
			thisScene.getLayer('2').visible = false
			thisScene.getLayer('3').visible = false

			thisScene.getLayer('红').visible = true
			thisScene.getLayer('白').visible = false
	}else if(shared.back == 3){
			thisScene.getLayer('00').visible = false
			thisScene.getLayer('纯色').visible = false
			thisScene.getLayer('可调整组合层').visible = false
			thisScene.getLayer('1').visible = false
			thisScene.getLayer('2').visible = false
			thisScene.getLayer('3').visible = false

			thisScene.getLayer('红').visible = false
			thisScene.getLayer('白').visible = true
	}

	return value;
}



export function applyUserProperties(changedUserProperties) {

	if(changedUserProperties.hasOwnProperty('back')){
		if(changedUserProperties.back == '路飞1'){
			shared.back = 1
		}else if(changedUserProperties.back == '路飞2'){
			shared.back = 2
		}else if(changedUserProperties.back == '路飞3'){
			shared.back = 3
		}
	}

}
```
