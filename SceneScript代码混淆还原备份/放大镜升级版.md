'use strict';
shared.a = false
let base_pos = {x:2560,y:720,z:0};
export function cursorMove(event) {

	var xzhou = input.cursorWorldPosition;
	thisScene.getLayer('映射顶部1').origin = {x:xzhou.x, y:base_pos.y, z:base_pos.z};
	thisScene.getLayer('映射边缘1').origin = {x:xzhou.x, y:base_pos.y, z:base_pos.z};
	thisScene.getLayer('被映底部1').origin = {x:xzhou.x, y:base_pos.y, z:base_pos.z};

	thisScene.getLayer('映射顶部2').origin = {x:base_pos.x, y:xzhou.y, z:base_pos.z};
	thisScene.getLayer('映射边缘2').origin = {x:base_pos.x, y:xzhou.y, z:base_pos.z};
	thisScene.getLayer('被映底部2').origin = {x:base_pos.x, y:xzhou.y, z:base_pos.z};

	thisScene.getLayer('方块顶部').origin = {x:xzhou.x, y:xzhou.y, z:base_pos.z};
	thisScene.getLayer('方块边缘').origin = {x:xzhou.x, y:xzhou.y, z:base_pos.z};
	thisScene.getLayer('方块底部').origin = {x:xzhou.x, y:xzhou.y, z:base_pos.z};

	thisScene.getLayer('圆圈顶部').origin = {x:xzhou.x, y:xzhou.y, z:base_pos.z};
	thisScene.getLayer('圆圈边缘').origin = {x:xzhou.x, y:xzhou.y, z:base_pos.z};
	thisScene.getLayer('圆圈底部').origin = {x:xzhou.x, y:xzhou.y, z:base_pos.z};

}
export function update(value) {    
	if(shared.a){
		thisScene.getLayer("映射顶部1").getEffect("你麻痹").visible = false
		thisScene.getLayer("映射边缘1").getEffect("你麻痹").visible = false
		thisScene.getLayer("映射顶部2").getEffect("你麻痹").visible = false
		thisScene.getLayer("映射边缘2").getEffect("你麻痹").visible = false
		thisScene.getLayer("方块顶部").getEffect("你麻痹").visible = false
		thisScene.getLayer("方块边缘").getEffect("你麻痹").visible = false
		thisScene.getLayer("圆圈顶部").getEffect("你麻痹").visible = false
		thisScene.getLayer("圆圈边缘").getEffect("你麻痹").visible = false
		thisScene.getLayer("kai").visible = true
		thisScene.getLayer("guan").visible = false
	}else{
		thisScene.getLayer("映射顶部1").getEffect("你麻痹").visible = true
		thisScene.getLayer("映射边缘1").getEffect("你麻痹").visible = true
		thisScene.getLayer("映射顶部2").getEffect("你麻痹").visible = true
		thisScene.getLayer("映射边缘2").getEffect("你麻痹").visible = true
		thisScene.getLayer("方块顶部").getEffect("你麻痹").visible = true
		thisScene.getLayer("方块边缘").getEffect("你麻痹").visible = true
		thisScene.getLayer("圆圈顶部").getEffect("你麻痹").visible = true
		thisScene.getLayer("圆圈边缘").getEffect("你麻痹").visible = true
		thisScene.getLayer("kai").visible = false
		thisScene.getLayer("guan").visible = true
	}
	return value;
}
export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('kaiguan')){
		shared.a=changedUserProperties.kaiguan;
	}
}
