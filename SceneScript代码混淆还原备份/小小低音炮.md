'use strict';
shared.roundmusiclogoswitch = true
shared.roundmusic = 0
shared.roundmusiclogo = 1
shared.backmusicswitch = true
shared.backmusic = 0
shared.backmusiclogo = 1

var fade_speed = 1;
fade_speed = 1 / fade_speed;
var ofg = 0
var timelogo = 0
var timemusic = 0
var timlogo = 0
var timmusic = 0

export function mediaPlaybackChanged(event) {
    if(event.state == 0){
		ofg = 0
	}else if(event.state == 1){
		ofg = 1
	}else if(event.state == 2){
		ofg = 2
	}else{
		ofg = 3
	}
}
//关闭 0    播放1   暂停2  
export function update(value) {
    if(shared.roundmusiclogoswitch){
		thisScene.getLayer('logo自定义音频').visible == true
		thisScene.getLayer('logo边音频').visible == true
		timelogo += engine.frametime * 0.3 ;
		timemusic += engine.frametime * 0.3 ;
		if(ofg == 0){
			shared.roundmusiclogo += timelogo
			if (shared.roundmusiclogo >= 1.0) {
				shared.roundmusiclogo = 1.0;
				timelogo = 0
			}
			shared.roundmusic -= timemusic
			if (shared.roundmusic <= 0.0) {
				shared.roundmusic = 0.0;
				timemusic = 0
			}
		}else if(ofg == 1){
			shared.roundmusic += timemusic
			if (shared.roundmusic >= 1.0) {
				shared.roundmusic = 1.0;
				timemusic = 0
			}
			shared.roundmusiclogo -= timelogo
			if (shared.roundmusiclogo <= 0.0) {
				shared.roundmusiclogo = 0.0;
				timelogo = 0
			}			
		}else if(ofg == 2){
			shared.roundmusiclogo += timelogo
			if (shared.roundmusiclogo >= 1.0) {
				shared.roundmusiclogo = 1.0;
				timelogo = 0
			}
			shared.roundmusic -= timemusic
			if (shared.roundmusic <= 0.0) {
				shared.roundmusic = 0.0;
				timemusic = 0
			}
		}
	}else if(shared.roundmusiclogoswitch == false){
		shared.roundmusic = 0
		shared.roundmusiclogo = 1
		thisScene.getLayer('logo自定义音频').visible == false
		thisScene.getLayer('logo边音频').visible == false
	}

	
    if(shared.backmusicswitch){
		timlogo += engine.frametime * 0.3 ;
		timmusic += engine.frametime * 0.3 ;
		if(ofg == 0){
			shared.backmusiclogo += timlogo
			if (shared.backmusiclogo >= 1.0) {
				shared.backmusiclogo = 1.0;
				timlogo = 0
			}
			shared.backmusic -= timmusic
			if (shared.backmusic <= 0.0) {
				shared.backmusic = 0.0;
				timmusic = 0
			}
		}else if(ofg == 1){
			shared.backmusic += timmusic
			if (shared.backmusic >= 1.0) {
				shared.backmusic = 1.0;
				timmusic = 0
			}
			shared.backmusiclogo -= timlogo
			if (shared.backmusiclogo <= 0.0) {
				shared.backmusiclogo = 0.0;
				timlogo = 0
			}			
		}else if(ofg == 2){
			shared.backmusiclogo += timlogo
			if (shared.backmusiclogo >= 1.0) {
				shared.backmusiclogo = 1.0;
				timlogo = 0
			}
			shared.backmusic -= timmusic
			if (shared.backmusic <= 0.0) {
				shared.backmusic = 0.0;
				timmusic = 0
			}
		}
	}else if(shared.backmusicswitch == false){
		shared.backmusic = 0
		shared.backmusiclogo = 1
	}



    return value
}

export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('newproperty6')){
		shared.roundmusiclogoswitch=changedUserProperties.newproperty6;
	}
	if(changedUserProperties.hasOwnProperty('newproperty13')){
		shared.backmusicswitch=changedUserProperties.newproperty13;
	}
}
