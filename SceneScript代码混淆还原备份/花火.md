'use strict';

shared = {
	//点击人物切换
	cickrw:1,
	//音乐
	music:false,
	//面具
	mj:1,
	//提示
	ts:true
}
var timer = 0
var toum = 1
export function update(value) {
	timer += engine.frametime;
	if(shared.ts){
		thisScene.getLayer('提示').visible = shared.ts
		thisScene.getLayer('提示').alpha = toum

		if (timer >= 3) {
			toum -= engine.frametime * 8;
			if(toum <= 0){
				thisScene.getLayer('提示').visible = false
			}
		}

	}else{
		thisScene.getLayer('提示').visible = shared.ts
	}
	return value;
}

export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('jiaose')){
		shared.cickrw=changedUserProperties.jiaose;
	}
	if(changedUserProperties.hasOwnProperty('ui_editor_properties_alert_tip')){
		shared.ts=changedUserProperties.ui_editor_properties_alert_tip;
	}
}

'use strict';
import * as WEMath from 'WEMath';

export var scriptProperties = createScriptProperties()
	.addSlider({
		name: 'speed',
		label: '变化速度' +'   '+ 'Rate of change',
		value: 10,
		min: 1,
		max: 100,
		integer: false
	})
	.finish();
// Please note: Do not remove this line or asset references may break.
/*options start*/
var fade_speed = 1;     //time to fade in (in seconds)
var basic_volume = 1; //basic volume (from 0 to 1)
/*options end*/

var state = 0;
var fade_counter = 0;
fade_speed = 1 / fade_speed;

var ofg = 0

var hover = true
var speed = 0
var time = 0, intervalTime = 0, oldTime = 0, n = 0 , frame = 60;

var timer1 = 0
var timer2 = 0

export function mediaPlaybackChanged(event) {
    if(event.state == 0){
		ofg = 0
	}else if(event.state == 1){
		ofg = 1
	}else if(event.state == 2){
		ofg = 2
	}else{
		ofg = 3
	}
}
//关闭 0    播放1   暂停2  
export function mediaThumbnailChanged(event) {
    // console.log(event.hasThumbnail)
    if(event.hasThumbnail){
        ofg = 1
    }else{
        ofg = 0
    }
}
export function update(value) {
    //获取帧率
	intervalTime += engine.frametime;
	n++;
	if (time - oldTime > 500){
		let fps = 1 / (intervalTime / n);
		frame = fps.toFixed(0);
		oldTime = time;
		intervalTime = n = 0;
	}
	time += engine.frametime * 1000;
	speed = (scriptProperties.speed / 100) * (60 / frame);
	
	if(ofg == 0){
		timer1 = 0
	}else if(ofg == 1){
		timer1 += engine.frametime;
        if (timer1 > 1) {
			
            value.y = WEMath.mix(value.y, 50, speed);
        }
        if (timer1 > 3.5) {
            value.y = WEMath.mix(value.y, 3000, speed);
        }
		if (timer1 > 4){
			value.y = -1350
			timer1 = 4
		}
	}else if(ofg == 2){
		timer1 = 0
	}
	// console.log(value.y)
    
    // return fade_counter * basic_volume;
	return value;
}
