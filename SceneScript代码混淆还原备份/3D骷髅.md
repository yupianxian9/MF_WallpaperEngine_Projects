        value: 0,
        min: -5,
        max: 5,
        integer: false

export function update(value) {
    value.z = scriptProperties.z
    if (scriptProperties.active) {
        // Calculate the cursor delta
        let cursorDelta = input.cursorWorldPosition.subtract(initPos);
        if (scriptProperties.reverseDirection) {
            cursorDelta = cursorDelta.multiply(-1);
        }

        cursorDelta = cursorDelta.multiply(scriptProperties.sensitivity);

        // Calculate target position based on cursor influence
        targetPos = initPos.copy();
        const dist = new Vec2(scriptProperties.distanceScale, scriptProperties.distanceScale).multiply(cursorDelta.length());
        const maxDist = new Vec2(scriptProperties.maxDistanceX, scriptProperties.maxDistanceY);
        cursorDelta = cursorDelta.normalize().multiply(minVec(maxDist, dist));
        targetPos = targetPos.add(cursorDelta);

        // Apply attraction effect
        let attractionDelta = input.cursorWorldPosition.subtract(targetPos);
        let attractionDistance = attractionDelta.length();

        if (attractionDistance < scriptProperties.attractionRadius) {
            let attractionStrength = (scriptProperties.attractionRadius - attractionDistance) / scriptProperties.attractionRadius * scriptProperties.attractionStrength;
            attractionDelta = attractionDelta.normalize().multiply(attractionStrength);
            targetPos = targetPos.add(attractionDelta);
        }

        // Apply smoothing
        currentPos = smoothVec(currentPos, targetPos, scriptProperties.smoothing);
        // console.log('位置' + 'x:' + value.x.toFixed(2) + ' y:' + value.y.toFixed(2) + ' z:' + value.z.toFixed(2));
        // Interpolate the position using high precision matrix operations
        return mixVecHighPrecision(value, currentPos, scriptProperties.delay);
    }
    return initPos;
}

export function init(value) {
    initPos = value;
    currentPos = value.copy();
    targetPos = initPos.copy();
    return value;
}

function minVec(a, b) {
    return new Vec2(Math.min(a.x, b.x), Math.min(a.y, b.y));
}

function mixVecHighPrecision(a, b, v) {
    return new Vec3(
        WEMath.mix(a.x, b.x, v),
        WEMath.mix(a.y, b.y, v),
        WEMath.mix(a.z, b.z, v) // Keep z-axis mixing for 3D scenes
    );
}

function smoothVec(a, b, smoothing) {
    return new Vec3(
        WEMath.mix(a.x, b.x, smoothing),
        WEMath.mix(a.y, b.y, smoothing),
        WEMath.mix(a.z, b.z, smoothing) // Keep z-axis smoothing for 3D scenes
    );
}

let smoothvalue;
var audioCache = [];
let animtime = 0;
let fadein = 0;

/**
 * Handles Cube idle animation and audio input
 */

export function update(value) {
	if (audioBuffer.average.reduce((a, b) => a + b) < 1 && engine.userProperties.idle) {
		idleProcessing();
		
	} else {
		fadein = 0;
		audioProcessing();
	}




}


/**
 * Incomming audio is being processed as normal. Smooth audio appends x samples into an array to find out the mean.
 *Audio  will be clamped from 0.2 to 1.1 to erase weak values.
 */
export function audioProcessing() {
	const scaleDelta = scriptProperties.maxvalue - scriptProperties.minvalue;

	for (var i = 0; i < cubeLayers.length; i++) {
		let audioDelta = audioBuffer.average[i] - smoothValue;
		smoothValue += audioDelta * Math.min(1.0, engine.frametime * scriptProperties.smoothing);

		if (scriptProperties.smoothAudio) {
			audioCache[i].unshift(audioBuffer.average[i]);//TODO: avg values in array return > 1...
			audioCache[i].length = scriptProperties.smoothrate;
			let audiomean = (audioCache[i].reduce((a, b) => (a + b)) / (audioCache[i].length));
			smoothvalue = normalize(0.2, 1, audiomean);
			if (i == 0) {

				shared.smoothbass = smoothvalue;
			}
			cubeLayers[i].scale = new Vec3(cubeLayers[i].scale.x, cubeLayers[i].scale.y, smoothvalue * scriptProperties.maxvalue * 0.05 + 0.03, scriptProperties.maxvalue);

		} else {
			cubeLayers[i].scale = new Vec3(cubeLayers[i].scale.x, cubeLayers[i].scale.y, initialScale.z * smoothValue * scaleDelta + scriptProperties.minvalue);
		}
		if (i == 0) {

			var audio = initialScale.z * smoothValue * scaleDelta + scriptProperties.minvalue;
			shared.audiodata = audio;
			shared.smoothbass = (scriptProperties.smoothAudio ? shared.smoothbass : audio);
		}
	}

}


function idleProcessing() {
	cubeLayers.forEach((element, index) => {
		let sideshift = 0;
		sideshift = (index / 4) || 0;
		element.scale = new Vec3(element.scale.x, element.scale.y, Math.max((Math.sin(animtime + sideshift + index * (Math.PI / 2)) / 4 + 0.33) * fadein, 0.03));
	});
	animtime += engine.frametime / 2;
	animtime >= Math.PI ? 0 : animtime;
	shared.smoothbass = 0;
	fadein = Math.min(1, fadein + 0.001);
}

export function normalize(x, y, z) {
	return Math.max(Math.min((z - x) / (y - x), 1), 0);
}


export function smootherstep(a, b, x) {
	x = clamp((x - a) / (b - a), 0., 1.);
	return x * x * x * (x * (x * 6. - 15.) + 10.);
}

function clamp(num, min, max) {
	return num <= min ? min : num >= max ? max : num;
}

/**
 * @param {Vec3} value
 */
export function init(value) {
	for (var i = 1; i < 65; i++) {
		cubeLayers.push(thisScene.getLayer(i.toString()));
		audioCache.push([0]);
	}
	initialScale = value;
}
