'use strict';
var c
var click = c
export function cursorClick(event) {
	if(click == 1){
		click = 2
	}else if(click == 2){
		click = 3
	}else if(click == 3){
		click = 1
	}
}
export function update(value) {
	click=click
	if(click == 1){
		thisScene.getLayer('腿1').getEffect('白丝').visible = false
		thisScene.getLayer('腿1').getEffect('黑丝').visible = false
		thisScene.getLayer('园空').visible = true
		thisScene.getLayer('白色').visible = false
		thisScene.getLayer('黑色').visible = false
	}else if(click == 2){
		thisScene.getLayer('腿1').getEffect('白丝').visible = true
		thisScene.getLayer('腿1').getEffect('黑丝').visible = false
		thisScene.getLayer('园空').visible = false
		thisScene.getLayer('白色').visible = true
		thisScene.getLayer('黑色').visible = false
	}else if(click == 3){
		thisScene.getLayer('腿1').getEffect('白丝').visible = false
		thisScene.getLayer('腿1').getEffect('黑丝').visible = true
		thisScene.getLayer('园空').visible = false
		thisScene.getLayer('白色').visible = false
		thisScene.getLayer('黑色').visible = true
	}
	return value;
}
export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('newproperty')){
		click = changedUserProperties.newproperty
		c = changedUserProperties.newproperty
	}
}
