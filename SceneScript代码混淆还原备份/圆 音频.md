'use strict';
var audioBuffer = engine.registerAudioBuffers(16);
var state
export function update(value) {
    state = !!audioBuffer.average.reduce((a, b) => a + b);
	thisScene.getLayer('音条1').visible = state
	thisScene.getLayer('音条2').visible = state
	thisScene.getLayer('音条3').visible != state
	thisScene.getLayer('音条4').visible != state
	// console.log(state)
	return value;
}

'use strict';

import * as WEMath from 'WEMath';

export var scriptProperties = createScriptProperties()
	.addSlider({
		name: 'speed',
		label: 'Speed',
		value: 0.04,
		min: 0.001,
		max: 10,
		integer: false
	})
	.addSlider({
		name: 'blinkAfterTimeout',
		label: 'Cursor Blink Duration',
		value: 0,
		min: 0,
		max: 100,
		integer: false
	})
	.finish();

let timer = 0;
let index = 0;
let newWord = "";
let deleteWord = false;
let queuedWord = "";
let blinkerOn = true;
let blinkEnabled = false;
let blinkAfterTimeoutTimer = 0;
let blinkTimeout = 1;
let blinkTimer = 0;

let trackName = ""; // 存储歌名

export function mediaPropertiesChanged(event) {
	if (event.title != null) {
		trackName = event.title;
	} else {
		trackName = "~";
	}
}

export function update(value) {
	let customText = trackName;

	if (typeof customText === 'string' || customText instanceof String) {
		queuedWord = customText;
	}

	value = handleBlinker(value);

	if (!deleteWord && timer >= scriptProperties.speed && index < newWord.length) {
		value = addLastChar(value);
		blinkTimer = 0;
	} else if (deleteWord && timer >= scriptProperties.speed && index > 0) {
		value = deleteLastChar(value);
	}

	if (queuedWord !== newWord) {
		deleteWord = true;
		if (index === 0 || queuedWord === value) {
			newWord = queuedWord;
			deleteWord = false;
		}
	}

	timer += engine.frametime;
	blinkTimer += engine.frametime;
	return value;
}

function handleBlinker(value) {
	if (scriptProperties.blinkAfterTimeout < blinkAfterTimeoutTimer) {
		blinkerOn = false;
		blinkAfterTimeoutTimer = 0;
	}
	if (blinkerOn && !blinkEnabled && blinkTimeout < blinkTimer && scriptProperties.blinkAfterTimeout > 0) {
		blinkEnabled = true;
		value += "|";
		blinkTimer = 0;
	} else if (blinkEnabled && blinkTimeout < blinkTimer) {
		value = value.slice(0, -1);
		blinkEnabled = false;
		blinkTimer = 0;
	}
	if (index === newWord.length) {
		blinkAfterTimeoutTimer += engine.frametime;
	}
	return value;
}

function addLastChar(value) {
	value = value + newWord[index];
	index = value.length;
	timer = 0;
	return value;
}

function deleteLastChar(value) {
	value = value.slice(0, -1);
	index = value.length;
	timer = 0;
	return value;
}

export function init(value) {
	blinkerOn = true;
	return "";
}
