var hover = true
var speed = 0
var time = 0, intervalTime = 0, oldTime = 0, n = 0 , frame = 60;

export function init(value) {
	shared.hb
	return value;
}
export function update(value) {
	//获取帧率
	intervalTime += engine.frametime;
	n++;
	if (time - oldTime > 500){
		let fps = 1 / (intervalTime / n);
		frame = fps.toFixed(0);
		oldTime = time;
		intervalTime = n = 0;
	}
	time += engine.frametime * 1000;
	speed = (scriptProperties.speed / 100) * (60 / frame);
	
	if (shared.hb == 1){
		//默认位置
		value = new Vec3(
			WEMath.mix(value.x, scriptProperties.ax, speed), 
			WEMath.mix(value.y, scriptProperties.ay, speed), 
			value.z, 
		);
		thisScene.getLayer('胶囊按钮').color = new Vec3(1, 1, 1)
	}else if(shared.hb == 2){	
		//点击位置
		value = new Vec3(
			WEMath.mix(value.x, scriptProperties.bx, speed), 
			WEMath.mix(value.y, scriptProperties.by, speed), 
			value.z, 
		);
		thisScene.getLayer('胶囊按钮').color = new Vec3(0, 0, 0)
	}
	return value;
}
export function cursorClick(event) {
	if(shared.hb == 1){
		shared.hb = 2
	}else{
		shared.hb = 1
	}
}

export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('newproperty')){
		shared.hb=changedUserProperties.newproperty;
	}
}
