```js
        targetColor = defaultColor.copy();

        targetColor = targetColor.multiply(1 / 255);
    }
}
```

```js
'use strict';

export var scriptProperties = createScriptProperties()
    .addColor({
        name: 'userColor',
        label: 'Custom Color',
        value: new Vec3(1, 1, 1) // 默认颜色
    })
    .finish();

const DURATION = 1; // 颜色过渡时间
let useAudioColor = false; // 是否使用音频颜色
let currentColor = scriptProperties.userColor;
let targetColor = scriptProperties.userColor;
let oldColor = scriptProperties.userColor;
let timer = DURATION;

export function update() {
    // 计算目标颜色
    targetColor = useAudioColor ? targetColor : scriptProperties.userColor;

    // 颜色过渡
    if (!currentColor.equals(targetColor)) {
        if (timer >= DURATION) {
            oldColor = currentColor;
            timer = 0;
        }

        const t = Math.min(timer / DURATION, 1.0);
        currentColor = new Vec3(
            oldColor.x + (targetColor.x - oldColor.x) * t,
            oldColor.y + (targetColor.y - oldColor.y) * t,
            oldColor.z + (targetColor.z - oldColor.z) * t
        );
        timer += engine.frametime;
    }

    return currentColor;
}

export function mediaThumbnailChanged(event) {
    if (!event.primaryColor || (event.primaryColor.x < 0.1 && event.primaryColor.y < 0.1 && event.primaryColor.z < 0.1)) {
        useAudioColor = false; // 关闭音频时回归自定义颜色
    } else {
        useAudioColor = true;
        targetColor = event.primaryColor; // 音频触发时更新颜色
        timer = 0;
    }
}
```

```js
let fadeOutTimer = 0;

export function update(value) {
    // console.log(1);
    const fadeSpeed = engine.frametime *2;

    if (shared.targetOfg == 1) {
        fadeOutTimer = 0; // 重置另一个计时器
        fadeInTimer += engine.frametime;
        if (fadeInTimer >= scriptProperties.fadeInDelay) {
            value += fadeSpeed;
            if (value >= 1.0) {
                value = 1.0;
            }
        }
    } else if (shared.targetOfg == 0 || shared.targetOfg == 2) {
        fadeInTimer = 0; // 重置另一个计时器
        fadeOutTimer += engine.frametime;
        if (fadeOutTimer >= scriptProperties.fadeOutDelay) {
            value -= fadeSpeed;
            if (value <= 0.0) {
                value = 0.0;
            }
        }
    }

    return value;
}
```
