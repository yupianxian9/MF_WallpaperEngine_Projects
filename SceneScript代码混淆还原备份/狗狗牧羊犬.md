```js
'use strict';
export function cursorEnter(event) {
	
	if(shared.lise == 2){
		thisScene.getLayer('1').visible = false
		thisScene.getLayer('2').visible = true
		thisScene.getLayer('g1').play();
	}else if(shared.lise == 3){
		thisScene.getLayer('1').visible = false
		thisScene.getLayer('3').visible = true
		thisScene.getLayer('g2').play();
	}else if(shared.lise == 4){
		thisScene.getLayer('1').visible = false
		thisScene.getLayer('3').visible = false
		thisScene.getLayer('4').visible = true
		thisScene.getLayer('g3').play();
	}

}
export function cursorLeave(event) {
	
	if(shared.lise == 2){
		thisScene.getLayer('1').visible = true
		thisScene.getLayer('2').visible = false
	}else if(shared.lise == 3){
		thisScene.getLayer('1').visible = true
		thisScene.getLayer('3').visible = false
	}else if(shared.lise == 4){
		thisScene.getLayer('1').visible = false
		thisScene.getLayer('3').visible = true
		thisScene.getLayer('4').visible = false
	}

}
```

```js
'use strict';

shared = {
	a:true,
	b:false,
	c:false,
	lise:2
}
export function update(value) {
	if(shared.lise == 2){
		thisScene.getLayer('圆点2s').visible = true
		thisScene.getLayer('圆点3s').visible = false
		thisScene.getLayer('圆点4s').visible = false
	}else if(shared.lise == 3){
		thisScene.getLayer('圆点2s').visible = false
		thisScene.getLayer('圆点3s').visible = true
		thisScene.getLayer('圆点4s').visible = false
	}else if(shared.lise == 4){
		thisScene.getLayer('圆点2s').visible = false
		thisScene.getLayer('圆点3s').visible = false
		thisScene.getLayer('圆点4s').visible = true
	}
	return value;
}

export function applyUserProperties(changedUserProperties) {
	if (changedUserProperties.hasOwnProperty('wapd')) {
		shared.lise = changedUserProperties.wapd;
	}
}
```
