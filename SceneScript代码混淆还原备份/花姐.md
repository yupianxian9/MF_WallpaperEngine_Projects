'use strict';
shared.img = true


export function update(value) {

	if(shared.img != true){
		thisScene.getLayer('纯色a0').visible = true
		thisScene.getLayer('纯色a1').visible = true
		thisScene.getLayer('纯色a2').visible = true
		thisScene.getLayer('纯色a3').visible = true
		thisScene.getLayer('纯色a4').visible = true
		thisScene.getLayer('纯色a5').visible = true
		thisScene.getLayer('纯色a6').visible = true
		thisScene.getLayer('开关2').color = thisScene.getLayer('音条s10').color
		
		thisScene.getLayer('方块1').visible = false
		thisScene.getLayer('方块2').visible = true
		thisScene.getLayer('方块3').visible = true


	}else{
		thisScene.getLayer('纯色a0').visible = false
		thisScene.getLayer('纯色a1').visible = false
		thisScene.getLayer('纯色a2').visible = false
		thisScene.getLayer('纯色a3').visible = false
		thisScene.getLayer('纯色a4').visible = false
		thisScene.getLayer('纯色a5').visible = false
		thisScene.getLayer('纯色a6').visible = false
		thisScene.getLayer('开关2').color = thisScene.getLayer('背景').color
		thisScene.getLayer('方块1').visible = true
		thisScene.getLayer('方块2').visible = false
		thisScene.getLayer('方块3').visible = false
	}
	return value;
}
export function applyUserProperties(changedUserProperties) {
	if(changedUserProperties.hasOwnProperty('newproperty3')){
		shared.img=changedUserProperties.newproperty3;
	}
}
