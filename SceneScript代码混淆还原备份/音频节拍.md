//点头1
'use strict';

export var scriptProperties = createScriptProperties()
	.addSlider({
		name: 's',
		label: 'New Slider',
		value: 10,
		min: 0,
		max: 100,
		integer: false
	})
	.addSlider({
		name: 'd',
		label: 'New Slider',
		value: 5,
		min: 0,
		max: 100,
		integer: false
	})
	.finish();
var audioBuffer = engine.registerAudioBuffers(16);
var state
var music
export function update(value) {
	state = !!audioBuffer.average.reduce((a, b) => a + b);
	if(state){
			music = audioBuffer.average[0] * scriptProperties.s
			if(music<=scriptProperties.d){
				value = scriptProperties.d
			}else{
				value = music
			}
	}else{
		value = 1
	}
	return value;
}
